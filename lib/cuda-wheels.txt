# A CUDA 13.2 toolkit from NVIDIA's pip wheels, installed by lib/cuda-home.sh into build/cuda-wheels (not
# into the venv: torch's cuda-toolkit[cudart] pins the venv's nvidia-cuda-runtime to 13.0.96, and nvcc, its
# crt headers, cicc and cccl must all be of one minor version for cccl's header check). 13.2 is the first
# pip crt whose rsqrt/rsqrtf declarations match glibc 2.42 and later (the 13.1.115 wheel's do not, and
# every DeepGEMM JIT build fails there). The kernels compiled with it run on the venv's 13.0 runtime.
nvidia-cuda-nvcc==13.2.86
nvidia-cuda-crt==13.2.86
nvidia-nvvm==13.2.86
nvidia-cuda-runtime==13.2.86
nvidia-cuda-cccl==13.2.86
